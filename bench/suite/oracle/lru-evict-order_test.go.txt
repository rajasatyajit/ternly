package simplelru

import "testing"

func TestOracleEvictsLeastRecentlyUsed(t *testing.T) {
	l, err := NewLRU[int, int](3, nil)
	if err != nil {
		t.Fatal(err)
	}
	for i := 1; i <= 3; i++ {
		l.Add(i, i)
	}
	l.Get(1) // 2 is now the least recently used
	if !l.Add(4, 4) {
		t.Fatal("adding to a full cache should evict")
	}
	if l.Contains(2) {
		t.Fatal("2 (least recently used) should have been evicted")
	}
	for _, k := range []int{1, 3, 4} {
		if !l.Contains(k) {
			t.Fatalf("%d should still be cached", k)
		}
	}
	if k, _, _ := l.GetOldest(); k != 3 {
		t.Fatalf("oldest = %d, want 3", k)
	}
	if k, _, ok := l.RemoveOldest(); !ok || k != 3 {
		t.Fatalf("RemoveOldest = %d %v, want 3", k, ok)
	}
}
