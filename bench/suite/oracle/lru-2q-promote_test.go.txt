package lru

import (
	"slices"
	"testing"
)

func TestOracle2QPromote(t *testing.T) {
	c, err := New2Q[int, int](4)
	if err != nil {
		t.Fatal(err)
	}
	c.Add(1, 1)
	c.Add(2, 2)
	if v, ok := c.Get(1); !ok || v != 1 {
		t.Fatalf("Get(1) = %v %v", v, ok)
	}
	if n := c.Len(); n != 2 {
		t.Fatalf("Len = %d, want 2", n)
	}
	keys := c.Keys()
	slices.Sort(keys)
	if !slices.Equal(keys, []int{1, 2}) {
		t.Fatalf("Keys = %v", keys)
	}
	c.Get(1)
	c.Remove(1)
	if c.Contains(1) {
		t.Fatal("1 still cached after Remove")
	}
	if n := c.Len(); n != 1 {
		t.Fatalf("Len after Remove = %d, want 1", n)
	}
}
