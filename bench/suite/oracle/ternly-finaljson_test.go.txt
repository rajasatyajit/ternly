package e2ejudge

import "testing"

func TestOracleFinalJSONLast(t *testing.T) {
	reply := "First a draft:\n```json\n{\"lines\": 1}\n```\nOn reflection:\n```json\n{\"lines\": 2}\n```\n"
	var v struct {
		Lines int `json:"lines"`
	}
	if err := FinalJSON(reply, &v); err != nil || v.Lines != 2 {
		t.Fatalf("FinalJSON = %+v, %v; want the last block (lines 2)", v, err)
	}
	var one struct {
		Lines int `json:"lines"`
	}
	if err := FinalJSON("```json\n{\"lines\": 7}\n```", &one); err != nil || one.Lines != 7 {
		t.Fatalf("one block: %+v %v", one, err)
	}
	if err := FinalJSON("no block", &one); err == nil {
		t.Fatal("no block must be an error")
	}
}
